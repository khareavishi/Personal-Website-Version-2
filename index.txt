    async function hashPassword(str) {
      try {
        const encoder = new TextEncoder();
        const data = encoder.encode(str);
        const hashBuffer = await crypto.subtle.digest(&#39;SHA-256&#39;, data);
        const hashArray = Array.from(new Uint8Array(hashBuffer));
        return hashArray.map(b =&gt; b.toString(16).padStart(2, &#39;0&#39;)).join(&#39;&#39;);
      } catch (e) {
        let h = 0;
        for (let i = 0; i &lt; str.length; i++) {
          h = ((h &lt;&lt; 5) - h) + str.charCodeAt(i);
          h |= 0;
        }
        return &#39;fallback_&#39; + Math.abs(h);
      }
    }

    // Universal photo uploader helper (works for both in-place slots and CMS)
    async function handlePhotoFile(file, photoId) {
      if (!file) return;
      const photo = (appState.photoArchive || []).find(p =&gt; p.id === photoId);
      if (!photo) return;

      showToast(&#39;Processing photo...&#39;);

      if (supabaseClient) {
        try {
          const fileExt = file.name.split(&#39;.&#39;).pop();
          const filePath = `${photoId}-${Date.now()}.${fileExt}`;
          showToast(&#39;Uploading to Supabase Storage CDN...&#39;);

          const { error: upErr } = await supabaseClient.storage
            .from(&#39;archive-photos&#39;)
            .upload(filePath, file, { upsert: true });

          if (!upErr) {
            const { data: pubUrlData } = supabaseClient.storage
              .from(&#39;archive-photos&#39;)
              .getPublicUrl(filePath);

            if (pubUrlData &amp;&amp; pubUrlData.publicUrl) {
              photo.url = pubUrlData.publicUrl;
              photo.status = &#39;Live on Supabase Storage&#39;;
              if (photoId === &#39;p1&#39;) appState.profile.heroImage.url = pubUrlData.publicUrl;
              await saveState();
              renderPublicSite();
              renderCmsStudio();
              showToast(&#39;Photo uploaded &amp; live on Supabase CDN&#39;);
              return;
            }
          }
        } catch (e) {
          console.warn(&#39;Supabase storage upload failed, falling back to local base64:&#39;, e);
        }
      }

      // Fallback: Read as Base64 data URL
      const reader = new FileReader();
      reader.onload = async (event) =&gt; {
        photo.url = event.target.result;
        photo.status = &#39;Custom Image Active&#39;;
        if (photoId === &#39;p1&#39;) appState.profile.heroImage.url = event.target.result;
        await saveState();
        renderPublicSite();
        renderCmsStudio();
        showToast(&#39;Photo updated for &#39; + photo.title);
      };
      reader.readAsDataURL(file);
    }

    function attachInPlacePhotoListeners() {
      // In-place file inputs
      document.querySelectorAll(&#39;.contextual-photo-input&#39;).forEach(inp =&gt; {
        inp.addEventListener(&#39;change&#39;, (e) =&gt; {
          const file = e.target.files[0];
          const photoId = inp.dataset.id;
          handlePhotoFile(file, photoId);
        });
      });

      // In-place resets
      document.querySelectorAll(&#39;.contextual-photo-reset&#39;).forEach(btn =&gt; {
        btn.addEventListener(&#39;click&#39;, () =&gt; {
          const photoId = btn.dataset.id;
          const photo = (appState.photoArchive || []).find(p =&gt; p.id === photoId);
          if (photo) {
            photo.url = &#39;&#39;;
            photo.status = &#39;Placeholder // Awaiting Upload&#39;;
            if (photoId === &#39;p1&#39;) appState.profile.heroImage.url = &#39;&#39;;
            saveState();
            renderPublicSite();
            renderCmsStudio();
            showToast(&#39;Photo reset to editorial placeholder&#39;);
          }
        });
      });

      // Lightbox triggers
      document.querySelectorAll(&#39;.photo-lightbox-trigger&#39;).forEach(trigger =&gt; {
        trigger.addEventListener(&#39;click&#39;, () =&gt; {
          const id = trigger.dataset.id;
          openLightbox(id);
        });
      });
    }

    function setupSupabaseFormListeners() {
      const form = document.getElementById('cmsSupabaseConfigForm');
      const urlInput = document.getElementById('cmsSupabaseUrl');
      const keyInput = document.getElementById('cmsSupabaseKey');
      const disconnectBtn = document.getElementById('cmsDisconnectSupabaseBtn');

      try {
        const raw = localStorage.getItem(SUPABASE_CONFIG_KEY);
        if (raw) {
          const cfg = JSON.parse(raw);
          if (cfg.url) urlInput.value = cfg.url;
          if (cfg.key) keyInput.value = cfg.key;
        }
      } catch (e) {}

      if (form) {
        form.addEventListener('submit', async (e) => {
          e.preventDefault();
          const url = urlInput.value.trim();
          const key = keyInput.value.trim();

          if (!url || !key) {
            showToast('Please enter both Supabase URL and Anon Key', 'error');
            return;
          }

          localStorage.setItem(SUPABASE_CONFIG_KEY, JSON.stringify({ url, key }));
          initSupabase();

          if (supabaseClient) {
            showToast('Testing Supabase Cloud connection...');
            try {
              const { data, error } = await supabaseClient
                .from('site_content')
                .select('data')
                .eq('id', 'main')
                .maybeSingle();

              if (!error && data && data.data) {
                appState = data.data;
                saveState(false);
                renderPublicSite();
                renderCmsStudio();
                showToast('Connected! Content loaded from Supabase Cloud');
              } else {
                await saveState(false);
                showToast('Connected! Initialized site_content on Supabase Cloud');
              }
            } catch (err) {
              showToast('Connected, but site_content check failed. Please run SQL script.', 'error');
            }
          }
        });
      }

      if (disconnectBtn) {
        disconnectBtn.addEventListener('click', () => {
          localStorage.removeItem(SUPABASE_CONFIG_KEY);
          supabaseClient = null;
          urlInput.value = '';
          keyInput.value = '';
          updateSupabaseStatus(false);
          showToast('Disconnected from Supabase. Reverted to Local Storage.');
        });
      }
    }

    document.addEventListener('DOMContentLoaded', () => {
      initSupabase();
      loadState();
      initTheme();
      startClock();
      renderPublicSite();
      setupSupabaseFormListeners();

      if (window.location.hash === '#admin') {
        showCmsStudio();
      }
    });
